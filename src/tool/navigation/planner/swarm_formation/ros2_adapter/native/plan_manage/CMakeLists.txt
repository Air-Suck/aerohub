cmake_minimum_required(VERSION 3.16)
project(ego_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(path_searching REQUIRED)
find_package(traj_opt REQUIRED)
find_package(traj_utils REQUIRED)
find_package(swarm_graph REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
include_directories(${PCL_INCLUDE_DIRS})
add_executable(ego_planner_node src/ego_planner_node.cpp src/ego_replan_fsm.cpp src/planner_manager.cpp)
target_include_directories(ego_planner_node PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ego_planner_node Eigen3::Eigen)
ament_target_dependencies(ego_planner_node rclcpp plan_env path_searching traj_opt traj_utils swarm_graph std_msgs nav_msgs sensor_msgs geometry_msgs visualization_msgs)
install(TARGETS ego_planner_node EXPORT export_ego_planner
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_ego_planner HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 rclcpp plan_env path_searching traj_opt traj_utils swarm_graph std_msgs nav_msgs sensor_msgs geometry_msgs visualization_msgs)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/ros2_proxy.py DESTINATION lib/${PROJECT_NAME})
ament_package()
