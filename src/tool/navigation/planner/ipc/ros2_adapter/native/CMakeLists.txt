cmake_minimum_required(VERSION 3.16)
project(ipc_native_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
find_package(OsqpEigen REQUIRED)
add_library(ipc_core core/astar.cpp core/local_astar.cpp core/mpc.cpp core/engine.cpp)
target_include_directories(ipc_core PUBLIC core ${PCL_INCLUDE_DIRS})
target_link_libraries(ipc_core PUBLIC fmt::fmt Eigen3::Eigen OsqpEigen::OsqpEigen ${PCL_LIBRARIES})
add_executable(ipc_core_smoke tests/core_smoke.cpp)
target_link_libraries(ipc_core_smoke PRIVATE ipc_core fmt::fmt)
enable_testing()
add_test(NAME ipc_core_smoke COMMAND ipc_core_smoke)
find_package(ament_cmake QUIET)
if(ament_cmake_FOUND)
  find_package(rclcpp REQUIRED)
  find_package(as2_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(uav_follow_reference_acro REQUIRED)
  add_executable(ipc_node src/node.cpp)
  target_link_libraries(ipc_node ipc_core)
  ament_target_dependencies(ipc_node rclcpp as2_msgs nav_msgs sensor_msgs std_msgs uav_follow_reference_acro)
  install(TARGETS ipc_node DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
add_executable(ipc_engine_smoke tests/engine_smoke.cpp)
target_link_libraries(ipc_engine_smoke PRIVATE ipc_core)
add_test(NAME ipc_engine_smoke COMMAND ipc_engine_smoke)
add_executable(ipc_corridor_pruning tests/corridor_pruning.cpp)
target_link_libraries(ipc_corridor_pruning PRIVATE ipc_core)
add_test(NAME ipc_corridor_pruning COMMAND ipc_corridor_pruning)
add_executable(ipc_three_dimensional_path tests/three_dimensional_path.cpp)
target_link_libraries(ipc_three_dimensional_path PRIVATE ipc_core)
add_test(NAME ipc_three_dimensional_path COMMAND ipc_three_dimensional_path)

add_executable(ipc_corridor_schedule tests/corridor_schedule.cpp)
target_link_libraries(ipc_corridor_schedule PRIVATE ipc_core)
add_test(NAME ipc_corridor_schedule COMMAND ipc_corridor_schedule)
