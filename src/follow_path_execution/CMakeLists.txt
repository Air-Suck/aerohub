cmake_minimum_required(VERSION 3.16)
project(uav_follow_path_execution)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(deps as2_behaviors_motion as2_behavior as2_motion_reference_handlers
  as2_core as2_msgs rclcpp rclcpp_action std_srvs uav_planner_interfaces)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
add_executable(follow_path_extended_node src/follow_path_extended.cpp)
target_include_directories(follow_path_extended_node PRIVATE include)
ament_target_dependencies(follow_path_extended_node ${deps})
# Upstream exports the FollowPath header but not its component CMake target.
find_library(FOLLOW_PATH_COMPONENT follow_path_component REQUIRED)
target_link_libraries(follow_path_extended_node ${FOLLOW_PATH_COMPONENT})
install(TARGETS follow_path_extended_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(timeline_test tests/timeline_test.cpp)
  target_include_directories(timeline_test PRIVATE include)
  target_compile_options(timeline_test PRIVATE -UNDEBUG)
  add_test(NAME timeline COMMAND timeline_test)
endif()
ament_package()
