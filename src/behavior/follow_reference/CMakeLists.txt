cmake_minimum_required(VERSION 3.16)
project(uav_follow_reference_acro)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(deps as2_behavior as2_behaviors_motion as2_motion_reference_handlers as2_core as2_msgs
  pluginlib rclcpp rclcpp_action std_srvs rclcpp_components std_msgs geometry_msgs builtin_interfaces)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/ControlSession.msg" "msg/ControlFrame.msg"
  DEPENDENCIES std_msgs geometry_msgs builtin_interfaces as2_msgs)
add_library(follow_reference_acro_plugin SHARED src/plugin.cpp)
target_include_directories(follow_reference_acro_plugin PRIVATE include)
ament_target_dependencies(follow_reference_acro_plugin ${deps})
rosidl_get_typesupport_target(cpp_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(follow_reference_acro_plugin "${cpp_typesupport}")
pluginlib_export_plugin_description_file(as2_behaviors_motion plugins.xml)
install(TARGETS follow_reference_acro_plugin LIBRARY DESTINATION lib)
add_executable(follow_reference_acro_node src/node.cpp)
ament_target_dependencies(follow_reference_acro_node ${deps})
find_library(FOLLOW_REFERENCE_COMPONENT follow_reference_component REQUIRED)
target_link_libraries(follow_reference_acro_node ${FOLLOW_REFERENCE_COMPONENT})
install(TARGETS follow_reference_acro_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(stream_gate_test tests/stream_gate_test.cpp)
  target_include_directories(stream_gate_test PRIVATE include)
  target_compile_options(stream_gate_test PRIVATE -UNDEBUG)
  add_test(NAME stream_gate COMMAND stream_gate_test)
  add_executable(plugin_load_test tests/plugin_load_test.cpp)
  ament_target_dependencies(plugin_load_test ${deps})
  install(TARGETS plugin_load_test DESTINATION lib/${PROJECT_NAME})
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
