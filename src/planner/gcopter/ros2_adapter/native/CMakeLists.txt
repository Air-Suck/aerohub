cmake_minimum_required(VERSION 3.16)
project(uav_gcopter_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Eigen3 REQUIRED)
find_package(ompl REQUIRED)
set(GCOPTER_UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../upstream/gcopter" CACHE PATH "Locked GCOPTER source")
# Replay the empty-obstacle compatibility fix without modifying the locked tree.
file(READ "${GCOPTER_UPSTREAM}/include/gcopter/sfc_gen.hpp" GCOPTER_SFC)
string(REPLACE "valid_pc[0].data()" "(valid_pc.empty() ? nullptr : valid_pc[0].data())" GCOPTER_SFC "${GCOPTER_SFC}")
# OMPL defaults to 1% of the whole map diagonal (~2.85m here). Bound
# segment validation to half a voxel instead of skipping thin obstacles.
string(REPLACE "si->setup();" "si->setStateValidityCheckingResolution(std::min(0.01, 0.5 * mapPtr->getScale() / space->getMaximumExtent()));\n        si->setup();" GCOPTER_SFC "${GCOPTER_SFC}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/compat/gcopter")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compat/gcopter/sfc_gen.hpp" "${GCOPTER_SFC}")
add_library(uav_gcopter_core STATIC src/core.cpp)
target_include_directories(uav_gcopter_core BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/compat" "${GCOPTER_UPSTREAM}/include/gcopter")
target_include_directories(uav_gcopter_core PUBLIC include "${GCOPTER_UPSTREAM}/include" ${OMPL_INCLUDE_DIRS})
target_link_libraries(uav_gcopter_core PUBLIC Eigen3::Eigen ${OMPL_LIBRARIES})
add_executable(gcopter_core_check tests/core_check.cpp)
target_link_libraries(gcopter_core_check PRIVATE uav_gcopter_core)
enable_testing()
add_test(NAME gcopter_core_check COMMAND gcopter_core_check)
install(TARGETS uav_gcopter_core gcopter_core_check ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY "${GCOPTER_UPSTREAM}/include/gcopter" DESTINATION include)
install(FILES "${GCOPTER_UPSTREAM}/../LICENSE" DESTINATION share/${PROJECT_NAME})
option(GCOPTER_WITH_ROS "Build the ROS 2 Function harness" OFF)
if(GCOPTER_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(uav_planner_interfaces REQUIRED)
  add_executable(gcopter_function src/node.cpp)
  target_link_libraries(gcopter_function uav_gcopter_core)
  ament_target_dependencies(gcopter_function rclcpp nav_msgs sensor_msgs std_srvs uav_planner_interfaces)
  install(TARGETS gcopter_function DESTINATION lib/${PROJECT_NAME})
  find_package(std_msgs REQUIRED)
  find_package(PCL REQUIRED COMPONENTS common io filters)
  find_package(pcl_conversions REQUIRED)
  find_package(yaml-cpp REQUIRED)
  file(GLOB GCOPTER_MAP_SOURCES vendor/rog_map/src/rog_map/*.cpp)
  add_library(gcopter_local_map STATIC ${GCOPTER_MAP_SOURCES} vendor/rog_map/include/rog_map/rog_map_core/raycaster.cpp)
  target_compile_features(gcopter_local_map PUBLIC cxx_std_17)
  target_compile_definitions(gcopter_local_map PUBLIC ORIGIN_AT_CORNER USE_ROS2 ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/vendor/rog_map/")
  target_include_directories(gcopter_local_map PUBLIC vendor/rog_map/include ${PCL_INCLUDE_DIRS})
  target_link_libraries(gcopter_local_map PUBLIC ${PCL_LIBRARIES} yaml-cpp)
  ament_target_dependencies(gcopter_local_map PUBLIC rclcpp pcl_conversions)
  add_executable(gcopter_local_function src/local_node.cpp)
  target_link_libraries(gcopter_local_function uav_gcopter_core gcopter_local_map)
  ament_target_dependencies(gcopter_local_function rclcpp nav_msgs sensor_msgs std_srvs std_msgs uav_planner_interfaces)
  install(TARGETS gcopter_local_function DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
  install(DIRECTORY vendor/rog_map/ DESTINATION share/${PROJECT_NAME}/vendor/rog_map)
  ament_export_include_directories(include)
  ament_export_libraries(uav_gcopter_core)
  ament_export_dependencies(Eigen3 ompl)
  ament_package()
endif()
