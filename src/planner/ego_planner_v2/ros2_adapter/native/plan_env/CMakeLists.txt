cmake_minimum_required(VERSION 3.16)
project(plan_env)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(traj_utils REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(pcl_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
include_directories(${PCL_INCLUDE_DIRS})
find_package(OpenCV REQUIRED)
include_directories(${OpenCV_INCLUDE_DIRS})
add_library(plan_env src/grid_map.cpp src/raycast.cpp)
target_include_directories(plan_env PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(plan_env Eigen3::Eigen)
# PointCloud2 conversion uses headers and PCL common. The exported pcl_conversions
# link list also includes PCL IO, which is unused here.
target_include_directories(plan_env PUBLIC ${pcl_conversions_INCLUDE_DIRS})
ament_target_dependencies(plan_env rclcpp traj_utils std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs cv_bridge message_filters pcl_msgs)
target_link_libraries(plan_env pcl_common ${OpenCV_LIBRARIES})
install(TARGETS plan_env EXPORT export_plan_env ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_plan_env HAS_LIBRARY_TARGET)
ament_export_include_directories(include ${pcl_conversions_INCLUDE_DIRS})
ament_export_dependencies(Eigen3 rclcpp traj_utils std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs cv_bridge message_filters pcl_msgs)
ament_package()
