cmake_minimum_required(VERSION 3.10)
project(Sophus LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
add_library(Sophus SHARED sophus/so2.cpp sophus/se2.cpp sophus/so3.cpp sophus/se3.cpp sophus/scso3.cpp sophus/sim3.cpp)
target_link_libraries(Sophus PUBLIC Eigen3::Eigen)
target_compile_features(Sophus PUBLIC cxx_std_11)
install(TARGETS Sophus LIBRARY DESTINATION lib)
install(DIRECTORY sophus/ DESTINATION include/sophus FILES_MATCHING PATTERN "*.h")
install(FILES SophusConfig.cmake DESTINATION lib/cmake/Sophus)
