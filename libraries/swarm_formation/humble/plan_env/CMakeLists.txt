cmake_minimum_required(VERSION 3.16)
project(plan_env)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(swarm_graph REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_sensor_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
include_directories(${PCL_INCLUDE_DIRS})
find_package(OpenCV REQUIRED)
include_directories(${OpenCV_INCLUDE_DIRS})
add_library(plan_env src/grid_map.cpp src/raycast.cpp)
target_include_directories(plan_env PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(plan_env Eigen3::Eigen)
ament_target_dependencies(plan_env rclcpp swarm_graph std_msgs sensor_msgs nav_msgs geometry_msgs visualization_msgs cv_bridge message_filters pcl_conversions)
target_link_libraries(plan_env ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
add_executable(navigation_bridge_node src/navigation_bridge.cpp)
ament_target_dependencies(navigation_bridge_node rclcpp nav_msgs sensor_msgs tf2_geometry_msgs tf2_sensor_msgs)
install(TARGETS navigation_bridge_node DESTINATION lib/${PROJECT_NAME})
option(SWARM_BUILD_VALIDATION "Build actual map input inspection" OFF)
if(SWARM_BUILD_VALIDATION)
  find_package(std_srvs REQUIRED)
  add_executable(map_probe_node tests/map_probe.cpp)
  target_link_libraries(map_probe_node plan_env)
  ament_target_dependencies(map_probe_node rclcpp sensor_msgs nav_msgs std_srvs swarm_graph)
  install(TARGETS map_probe_node DESTINATION lib/${PROJECT_NAME})
endif()
install(TARGETS plan_env EXPORT export_plan_env
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_plan_env HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 rclcpp swarm_graph std_msgs sensor_msgs nav_msgs geometry_msgs visualization_msgs cv_bridge message_filters pcl_conversions)
ament_package()
