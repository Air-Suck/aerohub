cmake_minimum_required(VERSION 3.16)
project(traj_opt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(path_searching REQUIRED)
find_package(traj_utils REQUIRED)
find_package(swarm_graph REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
include_directories(${PCL_INCLUDE_DIRS})
add_library(traj_opt src/poly_traj_optimizer.cpp)
target_include_directories(traj_opt PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(traj_opt Eigen3::Eigen)
ament_target_dependencies(traj_opt rclcpp plan_env path_searching traj_utils swarm_graph)
install(TARGETS traj_opt EXPORT export_traj_opt
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_traj_opt HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 rclcpp plan_env path_searching traj_utils swarm_graph)
option(SWARM_BUILD_VALIDATION "Build the shared ROS 1/2 differential benchmark" OFF)
if(SWARM_BUILD_VALIDATION)
  add_executable(optimizer_benchmark tests/optimizer_benchmark.cpp)
  target_compile_definitions(optimizer_benchmark PRIVATE SWARM_HUMBLE)
  target_link_libraries(optimizer_benchmark traj_opt)
  ament_target_dependencies(optimizer_benchmark rclcpp plan_env path_searching traj_utils swarm_graph)
  install(TARGETS optimizer_benchmark DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
